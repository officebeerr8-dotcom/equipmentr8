equipment
ระบบจัดการวัสดุอุปกรณ์คงคลัง

กรอก Supabase credentials เพื่อเชื่อมต่อฐานข้อมูล

SQL สำหรับสร้าง tables (รัน 1 ครั้งใน Supabase SQL Editor)

-- 1. วัสดุอุปกรณ์ CREATE TABLE materials ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), code text UNIQUE NOT NULL, name text NOT NULL, unit text NOT NULL, min_quantity numeric DEFAULT 0, image_url text, current_stock numeric DEFAULT 0, created_at timestamptz DEFAULT now(), updated_at timestamptz DEFAULT now() ); -- 2. รับเข้า docs CREATE TABLE receive_documents ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), doc_number text UNIQUE NOT NULL, note text, created_at timestamptz DEFAULT now() ); -- 3. รายการรับเข้า CREATE TABLE receive_items ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), doc_id uuid REFERENCES receive_documents(id) ON DELETE CASCADE, material_id uuid REFERENCES materials(id), quantity numeric NOT NULL, created_at timestamptz DEFAULT now() ); -- 4. เบิก-จ่าย docs CREATE TABLE withdrawal_documents ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), doc_number text UNIQUE NOT NULL, requester_name text NOT NULL, department text NOT NULL, note text, created_at timestamptz DEFAULT now() ); -- 5. รายการเบิก-จ่าย CREATE TABLE withdrawal_items ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), doc_id uuid REFERENCES withdrawal_documents(id) ON DELETE CASCADE, material_id uuid REFERENCES materials(id), quantity numeric NOT NULL, created_at timestamptz DEFAULT now() ); -- Trigger: อัปเดต stock เมื่อรับเข้า CREATE OR REPLACE FUNCTION update_stock_receive() RETURNS TRIGGER AS $$ BEGIN UPDATE materials SET current_stock = current_stock + NEW.quantity, updated_at = now() WHERE id = NEW.material_id; RETURN NEW; END; $$ LANGUAGE plpgsql; CREATE TRIGGER trg_receive AFTER INSERT ON receive_items FOR EACH ROW EXECUTE FUNCTION update_stock_receive(); -- Trigger: อัปเดต stock เมื่อเบิก-จ่าย CREATE OR REPLACE FUNCTION update_stock_withdraw() RETURNS TRIGGER AS $$ BEGIN UPDATE materials SET current_stock = current_stock - NEW.quantity, updated_at = now() WHERE id = NEW.material_id; RETURN NEW; END; $$ LANGUAGE plpgsql; CREATE TRIGGER trg_withdraw AFTER INSERT ON withdrawal_items FOR EACH ROW EXECUTE FUNCTION update_stock_withdraw(); -- RLS (allow all) ALTER TABLE materials ENABLE ROW LEVEL SECURITY; ALTER TABLE receive_documents ENABLE ROW LEVEL SECURITY; ALTER TABLE receive_items ENABLE ROW LEVEL SECURITY; ALTER TABLE withdrawal_documents ENABLE ROW LEVEL SECURITY; ALTER TABLE withdrawal_items ENABLE ROW LEVEL SECURITY; CREATE POLICY "all" ON materials FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "all" ON receive_documents FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "all" ON receive_items FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "all" ON withdrawal_documents FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "all" ON withdrawal_items FOR ALL USING (true) WITH CHECK (true);
